            statusMessage.textContent = message;

            statusMessage.style.display = 'block';
        }

        function hideStatus() { statusMessage.style.display = 'none'; }

        function openTab(evt, tabName) {
            document.querySelectorAll(".tab-content").forEach(tc => tc.style.display = "none");
            document.querySelectorAll(".tab-button").forEach(tb => tb.classList.remove("active"));
            document.getElementById(tabName).style.display = "block";
            evt.currentTarget.classList.add("active");
        }

        function formatTimestamp(timestamp) {
            if (timestamp == 0) return 'N/A';
            // As datas de formulário HTML são interpretadas como UTC, então o tratamento é simples
            const date = new Date(timestamp * 1000);
            return date.toLocaleDateString('pt-BR', { timeZone: 'UTC' });
        }

        function createBatteryCard(bateria) {

        async function connectWallet() {
            if (typeof window.ethereum === 'undefined') return alert("Por favor, instale o MetaMask!");
            try {
                provider = new ethers.providers.Web3Provider(window.ethereum);
                await provider.send("eth_requestAccounts", []);
                signer = provider.getSigner();
                const userAddress = await signer.getAddress();
                contract = new ethers.Contract(contractAddress, contractABI, signer);
                walletStatus.textContent = `Conectado: ${userAddress.substring(0, 6)}...${userAddress.substring(userAddress.length - 4)}`;
                connectButton.textContent = "Carteira Conectada";
                connectButton.disabled = true;
                document.getElementById('connect-section').style.backgroundColor = '#e9f5ff';
                dappContent.style.display = 'block';
                await listBatteries();
            } catch (error) {
                console.error("Erro ao conectar a carteira:", error);
                walletStatus.textContent = "Falha na conexão.";
            }
        }

        async function listBatteries() {
            if (!contract) return;
            showStatus('Buscando baterias registradas...', 'info');
            try {
                const batteries = await contract.listarBaterias();
                batteryListDiv.innerHTML = '';
                if (batteries.length === 0) {
                    batteryListDiv.innerHTML = '<p class="empty-state">Nenhuma bateria defeituosa encontrada.</p>';
                } else {
                    batteries.forEach(bateria => batteryListDiv.appendChild(createBatteryCard(bateria)));
                }
                hideStatus();
            } catch (error) {
                console.error("Erro ao listar baterias:", error);
                showStatus(`Erro ao buscar a lista: ${error.message}`, 'error');
            }
        }
        
        async function handleTransaction(form, action, ...args) {
            event.preventDefault();
            if (!contract) return;

            const actionMessages = {
                register: { start: 'Enviando transação para registro...', wait: 'Aguardando confirmação do registro...', success: 'Bateria registrada com sucesso!', error: 'Erro no registro:' },
                remove: { start: 'Enviando transação para remoção...', wait: 'Aguardando confirmação da remoção...', success: 'Bateria removida com sucesso!', error: 'Erro na remoção:' }
            };
            const messages = actionMessages[action];
            
            showStatus(messages.start, 'info');
            try {
                const contractMethod = action === 'register' ? contract.registrarBateria : contract.removerBateria;
                const tx = await contractMethod(...args);
                showStatus(messages.wait, 'info');
                await tx.wait();
                showStatus(messages.success, 'success');
                form.reset();
                await listBatteries();
            } catch (error) {
                console.error(`${messages.error} ${error}`);
                showStatus(`${messages.error} ${error.reason || error.message}`, 'error');
            }
        }

        registerForm.addEventListener('submit', (event) => {
            const tipo = document.getElementById('tipo').value;
            const uso = document.getElementById('uso').value;
            const serialNumber = document.getElementById('serialNumber').value;
            const dataProducaoStr = document.getElementById('dataProducao').value;
            const lote = document.getElementById('lote').value;
            // O campo de data do HTML já fornece uma data sem fuso horário (meia-noite UTC).
            // A conversão para timestamp é direta.
            const dataProducaoTimestamp = new Date(dataProducaoStr).getTime() / 1000;
            handleTransaction(registerForm, 'register', tipo, uso, serialNumber, dataProducaoTimestamp, lote);
        });

        removeForm.addEventListener('submit', (event) => {
            const serialNumber = document.getElementById('removeSerialNumber').value;
            handleTransaction(removeForm, 'remove', serialNumber);
        });
        
        async function consultarBateria(event) {
            event.preventDefault();
            if (!contract) return;
            const serialNumberToFind = document.getElementById('consultarSerialNumber').value;
            showStatus(`Consultando pelo número de série: ${serialNumberToFind}...`, 'info');
            consultaResultadoDiv.innerHTML = '';
            try {
                // A forma mais simples e que já funcionava: busca na lista completa.
                const batteries = await contract.listarBaterias();
                const resultado = batteries.find(b => b.serialNumber === serialNumberToFind);
                if (resultado) {
                    consultaResultadoDiv.appendChild(createBatteryCard(resultado));
                    hideStatus();
                } else {
                    consultaResultadoDiv.innerHTML = `<p class="empty-state">Nenhuma bateria encontrada com o número de série '${serialNumberToFind}'.</p>`;
                    hideStatus();
                }
            } catch (error) {
                console.error("Erro ao consultar bateria:", error);
                showStatus(`Erro na consulta: ${error.reason || error.message}`, 'error');
            }
        }
